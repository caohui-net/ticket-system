```typescript
async findOne(id: string): Promise<Ticket> {

// ✅ 列表查询（分页）
async findAll(filters: TicketFilter, pagination: PaginationDto) {
  const { page = 1, limit = 20 } = pagination;
  const skip = (page - 1) * limit;

  const [items, total] = await Promise.all([
    this.prisma.ticket.findMany({
      where: this.buildWhereClause(filters),
      skip,
      take: limit,
      orderBy: { createdAt: 'desc' },
      include: {
        creator: { select: { id: true, realName: true } },
        assignee: { select: { id: true, realName: true } },
      },
    }),
    this.prisma.ticket.count({
      where: this.buildWhereClause(filters),
    }),
  ]);

  return {
    items,
    pagination: {
      total,
      page,
      perPage: limit,
      totalPages: Math.ceil(total / limit),
    },
  };
}
```

### 6.2 查询构建规范

```typescript
private buildWhereClause(filters: TicketFilter): Prisma.TicketWhereInput {
  const where: Prisma.TicketWhereInput = ;

  // 状态筛选
  if (filters.status) {
    where.status = Array.isArray(filters.status)
      ? { in: filters.status }
      : filters.status;
  }

  // 优先级筛选
  if (filters.priority) {
    where.priority = filters.priority;
  }

  // 关键词搜索
  if (filters.keyword) {
    where.OR = [
      { title: { contains: filters.keyword, mode: 'insensitive' } },
      { description: { contains: filters.keyword, mode: 'insensitive' } },
    ];
  }

  // 时间范围
  if (filters.startDate || filters.endDate) {
    where.createdAt = {};
    if (filters.startDate) {
      where.createdAt.gte = new Date(filters.startDate);
    }
    if (filters.endDate) {
      where.createdAt.lte = new Date(filters.endDate);
    }
  }

  return where;
}
```

### 8.2 异常过滤器

```typescript
@Catch()
export class AllExceptionsFilter implements ExceptionFilter {
  constructor(private readonly logger: LoggerService) {}

  catch(exception: unknown, host: ArgumentsHost) {
    const ctx = host.switchToHttp();
    const response = ctx.getResponse();
    const request = ctx.getRequest();

    let status = HttpStatus.INTERNAL_SERVER_ERROR;
    let errorResponse: any = {
      success: false,
      error: {
        code: 'SYS_9001',
        message: '系统内部错误',
      },
      timestamp: new Date().toISOString(),
      path: request.url,
    };

    if (exception instanceof HttpException) {
      status = exception.getStatus();
      const exceptionResponse = exception.getResponse();

      if (typeof exceptionResponse === 'object') {
        errorResponse.error = exceptionResponse;
      } else {
        errorResponse.error.message = exceptionResponse;
      }
    } else if (exception instanceof Error) {
      errorResponse.error.message = exception.message;
    }

    // 记录错误日志
    this.logger.error(
      `${request.method} ${request.url}`,
      exception instanceof Error ? exception.stack : JSON.stringify(exception),
      'ExceptionFilter',
    );

    response.status(status).json(errorResponse);
  }
}
```
